  </div>

  <textarea id="comment" rows="3" maxlength="500" placeholder="Tulis komentar (opsional)"></textarea>
  <div class="center"><button id="submit" disabled>Kirim</button></div>
  <div id="msg"></div>

  <hr>
  <h3>Komentar Terbaru</h3>
  <div id="reviews">Memuat komentar...</div>
</div>

<script>
  const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbzdf0QpU0SFR8P1TUk5OqAlZNBGGscn7MW4DtshGYCM3AtBpEZswptz5AZx7kloMZ8_AA/exec';
  let selected = 0;
  const stars = document.querySelectorAll('.star');
  const btn = document.getElementById('submit');

  function paint(n) {
    stars.forEach(s => s.classList.toggle('active', Number(s.dataset.v) <= n));
  }

  stars.forEach(s => {
    s.addEventListener('mouseover', () => paint(Number(s.dataset.v)));
    s.addEventListener('mouseout', () => paint(selected));
    s.addEventListener('click', () => {
      selected = Number(s.dataset.v);
      paint(selected);
      btn.disabled = false;
    });
  });

  function renderReviews(list) {
    const box = document.getElementById('reviews');
    box.innerHTML = '';
    if (!list || !list.length) {
      box.textContent = 'Belum ada komentar.';
      return;
    }
    list.forEach(r => {
      const item = document.createElement('div');
      item.className = 'review';

      const top = document.createElement('div');
      top.className = 'top';
      top.textContent = '\u2605'.repeat(r.rating) + '\u2606'.repeat(5 - r.rating);

      const date = document.createElement('span');
      date.className = 'date';
      date.textContent = r.date;
      top.appendChild(date);

      const text = document.createElement('div');
      text.className = 'text';
      text.textContent = r.comment;

      item.appendChild(top);
      item.appendChild(text);
      box.appendChild(item);
    });
  }

  async function loadData() {
    try {
      const res = await fetch(SCRIPT_URL);
      const d = await res.json();
      document.getElementById('summary').textContent =
        d.count ? d.avg.toFixed(1) + ' / 5 dari ' + d.count + ' penilaian' : 'Belum ada rating';
      renderReviews(d.comments);
    } catch (e) {
      document.getElementById('summary').textContent = '';
      document.getElementById('reviews').textContent = '';
    }
  }

  btn.addEventListener('click', async () => {
    btn.disabled = true;
    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'text/plain' },
        body: JSON.stringify({
          rating: selected,
          comment: document.getElementById('comment').value
        })
      });
      document.getElementById('msg').textContent = 'Terima kasih atas rating Anda!';
      document.getElementById('comment').value = '';
      selected = 0;
      paint(0);
      setTimeout(loadData, 1500);
    } catch (e) {
      document.getElementById('msg').textContent = 'Gagal mengirim, coba lagi.';
      btn.disabled = false;
    }
  });

  loadData();
</script>
